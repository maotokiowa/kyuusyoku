function predictWebcam() { if (faceLandmarker && ui.presentationScreen.classList.contains('active')) { let startTimeMs = performance.now(); if (ui.video.readyState >= 2 && lastVideoTime !== ui.video.currentTime) { lastVideoTime = ui.video.currentTime; const results = faceLandmarker.detectForVideo(ui.video, startTimeMs); if (results.faceBlendshapes && results.faceBlendshapes.length > 0) { const shapes = results.faceBlendshapes[0].categories; // ★iPad用にブースト倍率を3.5倍〜4.5倍に引き上げ const left = ((shapes.find(s => s.categoryName === "eyeLookOutLeft")?.score || 0) + (shapes.find(s => s.categoryName === "eyeLookInRight")?.score || 0)) * 3.5; const right = ((shapes.find(s => s.categoryName === "eyeLookInLeft")?.score || 0) + (shapes.find(s => s.categoryName === "eyeLookOutRight")?.score || 0)) * 3.5; const up = ((shapes.find(s => s.categoryName === "eyeLookUpLeft")?.score || 0) + (shapes.find(s => s.categoryName === "eyeLookUpRight")?.score || 0)) * 4.5; // 上方向は特に弱いため強力補正 const down = ((shapes.find(s => s.categoryName === "eyeLookDownLeft")?.score || 0) + (shapes.find(s => s.categoryName === "eyeLookDownRight")?.score || 0)) * 3.5; const targetIndex = calculateGazeZone(left, right, up, down); updateDwellTime(targetIndex); } else { updateDwellTime(-1); } } } requestAnimationFrame(predictWebcam); } function calculateGazeZone(left, right, up, down) { const activeItems = defaultPresets.filter(p => selectedItems.includes(p.id)); const count = activeItems.length; let THRESHOLD = sensitivityThreshold; // どの方向にも目線が向いていない（極小の）場合のみ無効判定 if (left < THRESHOLD && right < THRESHOLD && up < THRESHOLD && down < THRESHOLD) { return -1; } let targetIndex = -1; if (count <= 2) { targetIndex = left > right ? 0 : 1; } else if (count <= 4) { let isRowUp = up > down; let isColLeft = left > right; if (isRowUp) { targetIndex = isColLeft ? 0 : 1; } else { targetIndex = isColLeft ? 2 : 3; } } else { // 5〜6項目の場合：左右の感度比較の差分条件を緩和（1.2倍の開きで判定） let isRowUp = up > down; let isColLeft = (left > right * 1.2); let isColRight = (right > left * 1.2); let col = 1; // 中央 if (isColLeft) col = 0; if (isColRight) col = 2; if (isRowUp) { targetIndex = col; } else { targetIndex = 3 + col; } } if (targetIndex >= count) { targetIndex = -1; } if (targetIndex !== -1) { const card = document.getElementById(`card-${targetIndex}`); if (card && card.classList.contains('disappeared')) { return -1; } } return targetIndex; }
